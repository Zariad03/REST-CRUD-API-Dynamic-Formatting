<body>

    <h1>My Favorite Books</h1>

    <p id="message">Loading books...</p>

    <table>
        <thead>
            <tr>
                <th>ID</th>
                <th>Title</th>
                <th>Author</th>
                <th>Rating</th>
                <th>Actions</th>
            </tr>
        </thead>

        <tbody id="book-list"></tbody>
    </table>

    <h2>Add a Book</h2>

    <form id="book-form">
        <input type="text" id="title"
               placeholder="Book Title" required>

        <input type="text" id="author"
               placeholder="Author" required>

        <input type="number" id="rating"
               placeholder="Rating (1-5)"
               min="1" max="5" required>

        <button type="submit">Add Book</button>
    </form>

    <script>
        async function loadBooks() {
            const message = document.getElementById("message");
            const bookList = document.getElementById("book-list");

            message.textContent = "Loading books...";

            try {
                const response = await fetch("/goodbooks");

                if (!response.ok) {
                    throw new Error("Unable to load books");
                }

                const books = await response.json();

                bookList.innerHTML = "";

                books.forEach(function(book) {
                    const row = document.createElement("tr");

                    if (book.rating >= 4) {
                        row.classList.add("high-rating");
                    }

                    const idCell = document.createElement("td");
                    const titleCell = document.createElement("td");
                    const authorCell = document.createElement("td");
                    const ratingCell = document.createElement("td");
                    const actionCell = document.createElement("td");

                    idCell.textContent = book.id;
                    titleCell.textContent = book.title;
                    authorCell.textContent = book.author;
                    ratingCell.textContent = book.rating;

                    const editButton = document.createElement("button");
                    editButton.textContent = "Edit";

                    editButton.onclick = function() {
                        editBook(book);
                    };

                    const deleteButton = document.createElement("button");
                    deleteButton.textContent = "Delete";

                    deleteButton.onclick = function() {
                        deleteBook(book.id);
                    };

                    actionCell.appendChild(editButton);
                    actionCell.appendChild(deleteButton);

                    row.appendChild(idCell);
                    row.appendChild(titleCell);
                    row.appendChild(authorCell);
                    row.appendChild(ratingCell);
                    row.appendChild(actionCell);

                    bookList.appendChild(row);
                });

                message.textContent = "Books loaded successfully.";

            } catch (error) {
                message.textContent = "Error loading books.";
            }
        }

        document.getElementById("book-form").addEventListener(
            "submit", async function(event) {

                event.preventDefault();

                const newBook = {
                    title: document.getElementById("title").value,
                    author: document.getElementById("author").value,
                    rating: Number(
                        document.getElementById("rating").value
                    )
                };

                try {
                    const response = await fetch("/goodbooks", {
                        method: "POST",
                        headers: {
                            "Content-Type": "application/json"
                        },
                        body: JSON.stringify(newBook)
                    });

                    if (!response.ok) {
                        throw new Error("Unable to add book");
                    }

                    document.getElementById("book-form").reset();

                    loadBooks();

                } catch (error) {
                    message.textContent = "Error adding book.";
                }
            }
        );

        async function editBook(book) {
            const title = prompt("Enter book title:", book.title);

            if (title === null) return;

            const author = prompt("Enter author:", book.author);

            if (author === null) return;

            const rating = Number(
                prompt("Enter rating from 1 to 5:", book.rating)
            );

            if (!Number.isInteger(rating) || rating < 1 || rating > 5) {
                document.getElementById("message").textContent =
                    "Rating must be between 1 and 5.";
                return;
            }

            try {
                const response = await fetch(
                    "/goodbooks/" + book.id,
                    {
                        method: "PUT",
                        headers: {
                            "Content-Type": "application/json"
                        },
                        body: JSON.stringify({
                            title: title,
                            author: author,
                            rating: rating
                        })
                    }
                );

                if (!response.ok) {
                    throw new Error("Unable to update book");
                }

                loadBooks();

            } catch (error) {
                document.getElementById("message").textContent =
                    "Error updating book.";
            }
        }

        async function deleteBook(bookId) {
            if (!confirm("Are you sure you want to delete this book?")) {
                return;
            }

            try {
                const response = await fetch(
                    "/goodbooks/" + bookId,
                    {
                        method: "DELETE"
                    }
                );

                if (!response.ok) {
                    throw new Error("Unable to delete book");
                }

                loadBooks();

            } catch (error) {
                document.getElementById("message").textContent =
                    "Error deleting book.";
            }
        }

        loadBooks();
    </script>
